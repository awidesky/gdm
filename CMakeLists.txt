    if(NOT DEFINED GDM_GLEW_VERSION)
        gdm_get_latest_version(

# CMAKE_PREFIX_PATH append for Windows 
if(WIN32)
    set(_gdm_prefix_paths "")

    # helper macro
    macro(gdm_add_prefix_if_exists path)
        if(EXISTS "${path}")
            list(APPEND _gdm_prefix_paths "${path}")
        endif()
    endmacro()

    if(GDM_WINDOW_BACKEND STREQUAL "glfw")
        gdm_add_prefix_if_exists("C:/Program Files/GLFW")
        gdm_add_prefix_if_exists("C:/Program Files (x86)/GLFW")
    elseif(GDM_WINDOW_BACKEND STREQUAL "freeglut")
        gdm_add_prefix_if_exists("C:/Program Files/freeglut")
        gdm_add_prefix_if_exists("C:/Program Files (x86)/freeglut")
    endif()
    
    if(GDM_USE_GLM)
        gdm_add_prefix_if_exists("C:/Program Files/glm")
        gdm_add_prefix_if_exists("C:/Program Files (x86)/glm")
    endif()
    
    if(GDM_GL_LOADER STREQUAL "glew" OR GDM_GL_LOADER STREQUAL "glew-glad")
        gdm_add_prefix_if_exists("C:/Program Files/GLEW")
        gdm_add_prefix_if_exists("C:/Program Files (x86)/GLEW")
    endif()

    # apply
    if(_gdm_prefix_paths)
        list(REMOVE_DUPLICATES _gdm_prefix_paths)
        string(JOIN ";" _gdm_prefix_string ${_gdm_prefix_paths})

        if(DEFINED CMAKE_PREFIX_PATH)
            set(CMAKE_PREFIX_PATH
                "${CMAKE_PREFIX_PATH};${_gdm_prefix_string}"
                CACHE PATH "Prefix path" FORCE
            )
        else()
            set(CMAKE_PREFIX_PATH
                "${_gdm_prefix_string}"
                CACHE PATH "Prefix path" FORCE
            )
        endif()
    endif()

# -----------------------
# Window backend (GLFW, FreeGLUT)
# -----------------------
if (GDM_WINDOW_BACKEND STREQUAL "none")
    add_library(gdm_window INTERFACE)
    add_library(gdm::window ALIAS gdm_window)
    list(APPEND _gdm_defs GDM_HAS_WINDOW_NONE)
elseif (GDM_WINDOW_BACKEND STREQUAL "glfw")
    # Derive git tag from version (strip trailing .0)
    set(GDM_GLFW_GIT_TAG ${GDM_GLFW_VERSION})
    string(REGEX REPLACE "\\.0$" "" GDM_GLFW_GIT_TAG "${GDM_GLFW_GIT_TAG}")

    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    if (GDM_GLFW_STATIC)
        set(GLFW_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
    else()
        set(GLFW_LIBRARY_TYPE SHARED CACHE STRING "" FORCE)
    endif()

    gdm_use_or_fetch_package(
        NAME glfw3
        VERSION ${GDM_GLFW_VERSION}
        GIT_REPOSITORY https://github.com/glfw/glfw
        GIT_TAG ${GDM_GLFW_GIT_TAG}
        ALIAS_TARGET gdm::window
        CANDIDATE_TARGETS glfw glfw3 glfw::glfw
    )

    if (MSVC)
        foreach(t glfw glfw3)
            if(TARGET ${t})
                get_target_property(_is_imported ${t} IMPORTED)

                if (NOT _is_imported)
                    target_compile_options(${t} PRIVATE "/utf-8")
                endif()
            endif()
        endforeach()
    endif()
    list(APPEND _gdm_defs GDM_HAS_GLFW)
elseif (GDM_WINDOW_BACKEND STREQUAL "freeglut")
    set(GDM_FREEGLUT_GIT_TAG "v${GDM_FREEGLUT_VERSION}")

    set(FREEGLUT_BUILD_DEMOS OFF CACHE BOOL "" FORCE)

    # freeglut's UNIX default backend is X11, and macOS ships no GLX headers
    # (its OpenGL is OpenGL.framework), and the native Cocoa backend is used.
    # That backend only exists from 3.8 on; a user who wants X11
    # on a Mac after all can still ask for it with -DFREEGLUT_COCOA=OFF.
    if(APPLE AND GDM_FREEGLUT_VERSION VERSION_GREATER_EQUAL 3.8 AND NOT DEFINED FREEGLUT_COCOA)
        set(FREEGLUT_COCOA ON CACHE BOOL "Use freeglut's native Cocoa backend (macOS)")
    endif()

    set(_gdm_freeglut_targets
        freeglut freeglut_static freeglut::freeglut freeglut::freeglut_static)

    gdm_use_or_fetch_package(
        NAME freeglut
        VERSION ${GDM_FREEGLUT_VERSION}
        GIT_REPOSITORY https://github.com/freeglut/freeglut
        GIT_TAG ${GDM_FREEGLUT_GIT_TAG}
        ALIAS_TARGET gdm::window
        CANDIDATE_TARGETS ${_gdm_freeglut_targets}
    )

    # freeglut declares no target_include_directories(); it only uses the legacy
    # directory-scope INCLUDE_DIRECTORIES(), which never reaches to the consumer.
    # So a freeglut built from the sources offers no path to GL/freeglut.h
    # A package found by find_package does carry its own include directory.
    foreach(_gdm_fg_target IN LISTS _gdm_freeglut_targets)
        if(TARGET ${_gdm_fg_target})
            get_target_property(_gdm_fg_includes ${_gdm_fg_target} INTERFACE_INCLUDE_DIRECTORIES)
            if(NOT _gdm_fg_includes)
                set(_gdm_fg_include_dir
                    "${GDM_EXTERNAL_DIR}/freeglut-${GDM_FREEGLUT_VERSION}/include")
                if(EXISTS "${_gdm_fg_include_dir}")
                    target_include_directories(${_gdm_fg_target} INTERFACE "${_gdm_fg_include_dir}")
                    message(STATUS
                        "[${PROJECT_NAME}] ${_gdm_fg_target} exposes no include "
                        "directory (freeglut's own CMake sets none); adding ${_gdm_fg_include_dir}"
                    )
                endif()
                unset(_gdm_fg_include_dir)
            endif()
            unset(_gdm_fg_includes)
        endif()
    endforeach()
    unset(_gdm_freeglut_targets)
    list(APPEND _gdm_defs GDM_HAS_FREEGLUT)
else()
    message(FATAL_ERROR "[GDM] Unsupported GDM_WINDOW_BACKEND: ${GDM_WINDOW_BACKEND}")
endif()

# -----------------------
# OpenGL loader (GLAD, GLEW)
# -----------------------

# glew-cmake <= 2.3.1 on Apple it unconditionally links the long-dead AGL framework.
# Upstream fixed this in PR #85 (USE_AGL option, default OFF), but that is not in
# a released tag yet - see https://github.com/Perlmint/glew-cmake/issues/84 .
if (APPLE AND (GDM_GL_LOADER STREQUAL "glew" OR GDM_GL_LOADER STREQUAL "glew-glad"))
    set(AGL_LIBRARY "" CACHE FILEPATH
        "AGL is deprecated and unused; skipped to work around glew-cmake linking it (issue #84)" FORCE)
endif()

if (GDM_GL_LOADER STREQUAL "none")
    add_library(gdm_loader INTERFACE)
    add_library(gdm::loader ALIAS gdm_loader)
    list(APPEND _gdm_defs GDM_HAS_LOADER_NONE)
elseif (GDM_GL_LOADER STREQUAL "glad")
    gdm_fetch_glad(
        OUT_VAR _glad_debug_dir
        BASE_DIR ${GDM_EXTERNAL_DIR}
        LANG "gl"
        APIS "${GDM_GLAD_API}"
        PROFILES "${GDM_GLAD_PROFILE}"
        OPTIONS DEBUG
        EXTENSIONS ${GDM_GLAD_EXTENSION}
    )

    gdm_fetch_glad(
        OUT_VAR _glad_release_dir
        BASE_DIR ${GDM_EXTERNAL_DIR}
        LANG "gl"
        APIS "${GDM_GLAD_API}"
        PROFILES "${GDM_GLAD_PROFILE}"
        EXTENSIONS ${GDM_GLAD_EXTENSION}
    )

    add_library(gdm_glad_debug STATIC
        "${_glad_debug_dir}/src/gl.c"
    )
    target_include_directories(gdm_glad_debug PUBLIC
        "${_glad_debug_dir}/include"
    )

    add_library(gdm_glad_release STATIC
        "${_glad_release_dir}/src/gl.c"
    )
    target_include_directories(gdm_glad_release PUBLIC
        "${_glad_release_dir}/include"
    )

    add_library(gdm_loader INTERFACE)
    add_library(gdm::loader ALIAS gdm_loader)
    add_dependencies(gdm_loader gdm_glad_debug gdm_glad_release)
    target_link_libraries(gdm_loader INTERFACE
        $<${_gdm_dbg}:gdm_glad_debug>
        $<$<NOT:${_gdm_dbg}>:gdm_glad_release>
    )
    list(APPEND _gdm_defs GDM_HAS_GLAD)

elseif (GDM_GL_LOADER STREQUAL "glew")
    if (GDM_WINDOW_BACKEND STREQUAL "none")
        message(FATAL_ERROR "[GDM] GDM_GL_LOADER=glew requires GDM_WINDOW_BACKEND=glfw or freeglut.")
    endif()

    # Prefer a system GLEW if actually provides the requested variant.
    # GLEW's own config package hard-errors when asked for a variant it does not ship 
    # (e.g. Homebrew installs shared-only).
    # So we leave the GLEW_USE_STATIC_LIBS unset, find whichever variant exists,
    # and pick the one matching GDM_GLEW_STATIC ourselves.
    find_package(GLEW QUIET)

    set(_gdm_glew_target "")
    if (TARGET GLEW::GLEW)
        if (GDM_GLEW_STATIC AND TARGET GLEW::glew_s)
            set(_gdm_glew_target GLEW::glew_s)
        elseif (NOT GDM_GLEW_STATIC AND TARGET GLEW::glew)
            set(_gdm_glew_target GLEW::glew)
        endif()
    endif()

    if (_gdm_glew_target)
        message(STATUS "[GDM] Found installed GLEW target: ${_gdm_glew_target}")
    else()
        if (TARGET GLEW::GLEW)
            message(STATUS
                "[GDM] GDM_GLEW_STATIC is ${GDM_GLEW_STATIC} but installed GLEW does not provide that build type. Falling back to glew-cmake."
            )
        endif()
        # Fallback: build the requested variant from source with glew-cmake.
        set(GDM_GLEW_GIT_TAG "glew-cmake-${GDM_GLEW_VERSION}")

        # Configure glew-cmake to build only libs (no glewinfo/visualinfo)
        set(ONLY_LIBS ON CACHE BOOL "" FORCE)

        if (GDM_GLEW_STATIC)
            set(glew-cmake_BUILD_SHARED OFF CACHE BOOL "" FORCE)
            set(glew-cmake_BUILD_STATIC ON CACHE BOOL "" FORCE)
        else()
            set(glew-cmake_BUILD_SHARED ON CACHE BOOL "" FORCE)
            set(glew-cmake_BUILD_STATIC OFF CACHE BOOL "" FORCE)
        endif()

        # Link through a private alias: the probe above may already have created
        # the public GLEW::GLEW target, which would stop gdm_use_or_fetch_package
        # from aliasing the freshly built one.
        gdm_use_or_fetch_package(
            NAME glew-cmake
            VERSION ${GDM_GLEW_VERSION}
            GIT_REPOSITORY https://github.com/Perlmint/glew-cmake
            GIT_TAG ${GDM_GLEW_GIT_TAG}
            ALIAS_TARGET gdm::glew
            CANDIDATE_TARGETS libglew_static libglew_shared
        )
        set(_gdm_glew_target gdm::glew)
    endif()

    add_library(gdm_loader INTERFACE)
    add_library(gdm::loader ALIAS gdm_loader)
    target_link_libraries(gdm_loader INTERFACE ${_gdm_glew_target})
    unset(_gdm_glew_target)
    list(APPEND _gdm_defs GDM_HAS_GLEW)

elseif (GDM_GL_LOADER STREQUAL "glew-glad")
    gdm_fetch_glad_allEXT(
        OUT_VAR _glad_allex_debug_dir
        BASE_DIR ${GDM_EXTERNAL_DIR}
        GLAD_OPTIONS DEBUG
    )

    gdm_fetch_glad_allEXT(
        OUT_VAR _glad_allex_release_dir
        BASE_DIR ${GDM_EXTERNAL_DIR}
    )

    add_library(gdm_glad_allext_debug STATIC
        "${_glad_allex_debug_dir}/src/gl.c"
    )
    target_include_directories(gdm_glad_allext_debug PUBLIC
        "${_glad_allex_debug_dir}/include"
    )

    add_library(gdm_glad_allext_release STATIC
        "${_glad_allex_release_dir}/src/gl.c"
    )
    target_include_directories(gdm_glad_allext_release PUBLIC
        "${_glad_allex_release_dir}/include"
    )

    # GLEW (always fetch from glew-cmake for glew-glad)
    set(GDM_GLEW_GIT_TAG "glew-cmake-${GDM_GLEW_VERSION}")

    # Configure glew-cmake to build only libs (no glewinfo/visualinfo)
    set(ONLY_LIBS ON CACHE BOOL "" FORCE)

    if (GDM_GLEW_STATIC)
        set(glew-cmake_BUILD_SHARED OFF CACHE BOOL "" FORCE)
        set(glew-cmake_BUILD_STATIC ON CACHE BOOL "" FORCE)
        set(_gdm_glewglad_glew_target libglew_static)
    else()
        set(glew-cmake_BUILD_SHARED ON CACHE BOOL "" FORCE)
        set(glew-cmake_BUILD_STATIC OFF CACHE BOOL "" FORCE)
        set(_gdm_glewglad_glew_target libglew_shared)
    endif()

    gdm_use_or_fetch_package(
        NAME glew-cmake
        VERSION ${GDM_GLEW_VERSION}
        GIT_REPOSITORY https://github.com/Perlmint/glew-cmake
        GIT_TAG ${GDM_GLEW_GIT_TAG}
        CANDIDATE_TARGETS ${_gdm_glewglad_glew_target}
    )
    if (NOT TARGET ${_gdm_glewglad_glew_target})
        message(FATAL_ERROR
            "[GDM] Failed to resolve glew-cmake target '${_gdm_glewglad_glew_target}' for glew-glad.")
    endif()


    # Find the directory that contains GL/glew.h
    set(_gdm_real_glew_root "${GDM_EXTERNAL_DIR}/glew-cmake-${GDM_GLEW_VERSION}/include")
    if (NOT _gdm_real_glew_root)
        message(FATAL_ERROR "[GDM] Could not locate real GL/glew.h in ${_gdm_real_glew_root}")
    endif()

    # Generate glew-glad wrapper header (fake glew.h)
    set(_gdm_real_glew_h "${_gdm_real_glew_root}/GL/glew.h")

    set(_gdm_glewglad_gen_root "${CMAKE_BINARY_DIR}/_gdm_deps/glew_glad")
    set(_gdm_glewglad_gen_inc  "${_gdm_glewglad_gen_root}/include")
    set(_gdm_fake_glew_h "${_gdm_glewglad_gen_inc}/GL/glew.h")

    # Skip header generation if it already exists
    if (NOT EXISTS "${_gdm_fake_glew_h}")
        file(MAKE_DIRECTORY "${_gdm_glewglad_gen_inc}/GL")
        file(STRINGS "${_gdm_real_glew_h}" _gdm_glew_h_lines)
        message(DEBUG "glew-glad header generation dir : ${_gdm_glewglad_gen_inc}")

        set(_gdm_strip_define_count 0)
        set(_gdm_strip_typedef_count 0)
        set(_gdm_strip_export_count 0)

        set(_gdm_fake_glew_h_text [=[/* Auto-generated by GDM: GLEW interface, but gl function loading by GLAD */
#ifndef GDM_GLEW_GLAD_GLEW_H
#define GDM_GLEW_GLAD_GLEW_H
]=])
        foreach(line IN LISTS _gdm_glew_h_lines)
            # Strip: #define glXxx GLEW_GET_FUN(...)
            if(line MATCHES "^[ \t]*#[ \t]*define[ \t]+gl[A-Za-z0-9_]+[ \t]+GLEW_GET_FUN[ \t]*[(]")
                math(EXPR _gdm_strip_define_count "${_gdm_strip_define_count} + 1")
                continue()
            endif()
            # Strip: typedef ... PFNGLXXXPROC ... (opening line of typedef, may be multi-line)
            if(line MATCHES "^[ \t]*typedef[ \t]+[^;]*[(]GLAPIENTRY[ \t]*[*][ \t]*PFNGL[A-Z0-9_]+PROC[)][ \t]*[(]")
                math(EXPR _gdm_strip_typedef_count "${_gdm_strip_typedef_count} + 1")
                continue()
            endif()
            # Strip: GLEW_FUN_EXPORT PFNGLXXXPROC __glewXxx;
            if(line MATCHES "^[ \t]*GLEW_FUN_EXPORT[ \t]+PFNGL[A-Z0-9_]+[ \t]+__glew[A-Za-z0-9_]+[ \t]*;")
                math(EXPR _gdm_strip_export_count "${_gdm_strip_export_count} + 1")
                continue()
            endif()

            # Strip: GLhandleARB must be void * in apple, unsigned int in else.
            # but glew.h does not do that.
            if(line MATCHES "^[ \t]*typedef[ \t]+unsigned[ \t]+int[ \t]+GLhandleARB[ \t]*;")
                continue()
            endif()

            string(APPEND _gdm_fake_glew_h_text "${line}\n")
        endforeach()
        string(APPEND _gdm_fake_glew_h_text [=[

// if not undef, glew.h will complain that another gl header is included.
#undef __gl_h_
#undef __glext_h_
// glad define some values like 0x000A, and glew does 0xA. so, undef or we get a warning
#if defined(_MSC_VER)
	#pragma warning(push)
	#pragma warning(disable : 4005)
	#include <glad/gl.h>
	#pragma warning(pop)
#elif defined(__clang__)
	#pragma clang diagnostic push
	#pragma clang diagnostic ignored "-Wmacro-redefined"
	#include <glad/gl.h>
	#pragma clang diagnostic pop
#elif defined(__GNUC__)
	#pragma GCC system_header
	#include <glad/gl.h>
#else
	#include <glad/gl.h>
#endif

#ifdef __cplusplus
extern "C" {
#endif
GLEWAPI GLenum GLEWAPIENTRY gdm_glew_glad_glewInit(void);
#ifdef __cplusplus
}
#endif
#define glewInit gdm_glew_glad_glewInit
#endif //GDM_GLEW_GLAD_GLEW_H
]=])
        file(WRITE "${_gdm_fake_glew_h}" "${_gdm_fake_glew_h_text}")
        unset(_gdm_fake_glew_h_text)
        unset(_gdm_glew_h_lines)
        message(STATUS
            "[GDM] glew-glad wrapper: "
            "removed ${_gdm_strip_define_count} gl* functions, "
            "${_gdm_strip_typedef_count} PFNGL typedefs, "
            "${_gdm_strip_export_count} GLEW_FUN_EXPORT declarations"
        )
    else()
        message(DEBUG "glew-glad wrapper header already exists, skipping generation")
    endif()

    # add glew wrapper front target
    add_library(gdm_glew_glad STATIC
        ${CMAKE_CURRENT_SOURCE_DIR}/src/glewToGlad.cpp
    )

    target_include_directories(gdm_glew_glad
        PUBLIC ${_gdm_glewglad_gen_inc}
    )
    unset(_gdm_real_glew_root)
    unset(_gdm_glewglad_gen_inc)
    unset(_gdm_glewglad_gen_root)

    add_library(gdm_glewglad_glad INTERFACE)
    add_dependencies(gdm_glewglad_glad gdm_glad_allext_debug gdm_glad_allext_release)
    target_link_libraries(gdm_glewglad_glad INTERFACE
        $<${_gdm_dbg}:gdm_glad_allext_debug>
        $<$<NOT:${_gdm_dbg}>:gdm_glad_allext_release>
    )
    target_link_libraries(gdm_glew_glad
        PUBLIC
            gdm_glewglad_glad
        PRIVATE
            gdm::window
            ${_gdm_glewglad_glew_target}
    )

    add_library(gdm_loader INTERFACE)
    add_library(gdm::loader ALIAS gdm_loader)
    target_link_libraries(gdm_loader INTERFACE gdm_glew_glad)

    list(APPEND _gdm_defs GDM_HAS_GLEW)
    list(APPEND _gdm_defs GDM_HAS_GLAD)
    list(APPEND _gdm_defs GDM_HAS_GLEW_GLAD)

    set_target_properties(gdm_glad_allext_debug   PROPERTIES FOLDER "GLEW_GLAD")
    set_target_properties(gdm_glad_allext_release PROPERTIES FOLDER "GLEW_GLAD")
    set_target_properties(${_gdm_glewglad_glew_target} PROPERTIES FOLDER "GLEW_GLAD")
    set_target_properties(gdm_glew_glad           PROPERTIES FOLDER "GLEW_GLAD")
    unset(_gdm_glewglad_glew_target)

else()
    message(FATAL_ERROR "[GDM] Unsupported GDM_GL_LOADER: ${GDM_GL_LOADER}")
endif()

if (TARGET gdm_glad_debug)
    set_target_properties(gdm_glad_debug PROPERTIES FOLDER "GLAD")
endif()
if (TARGET gdm_glad_release)
    set_target_properties(gdm_glad_release PROPERTIES FOLDER "GLAD")
